/*
 * EMAC1: H313 internal RMII EPHY via mainline dwmac-sun8i driver.
 *
 * The DTSI base node uses compatible "allwinner,sunxi-gmac" (BSP driver).
 * We override key properties so the mainline dwmac-sun8i driver binds:
 *   - compatible: triggers dwmac-sun8i binding (emac_variant_h313)
 *   - syscon = <&syscon 1>: EMAC1 clock register at SYSCON+0x34
 *   - clocks/clock-names: "stmmaceth" = CLK_BUS_EMAC1 (MAC bus clock),
 *     "ephy-25m" = CLK_EMAC_25M (25MHz reference for AC300 EPHY)
 *
 * The AC300 co-packaged EPHY (MDIO addr 0) is initialized via direct MII
 * register writes in dwmac-sun8i.c (internal_ephy=true path). No I2C MFD
 * driver is needed — AC300 uses MDIO only, unlike AC200 on H6/H616/H618.
 * tx/rx delays match Android BSP. [MCD125-DTS-v18-ac300-mdio-init]
 */
&emac1 {
	compatible = "allwinner,sun50i-h313-emac";
	syscon = <&syscon 1>;
	interrupt-names = "macirq";
	clocks = <&ccu CLK_BUS_EMAC1>, <&ccu CLK_EMAC_25M>;
	clock-names = "stmmaceth", "ephy-25m";
	phy-mode = "rmii";
	phy-handle = <&rmii_phy>;
	phy-supply = <&reg_aldo1>;
	allwinner,rx-delay-ps = <3100>;
	allwinner,tx-delay-ps = <700>;
	status = "okay";
};

&mdio1 {
	compatible = "snps,dwmac-mdio";
	rmii_phy: ethernet-phy@0 {
		/* AC300 co-packaged EPHY at MDIO addr 0.
		 * Initialized via MII writes in dwmac-sun8i probe (internal_ephy
		 * path). No I2C or MFD driver needed. [MCD125-DTS-v18] */
		reg = <0>;
	};
};
